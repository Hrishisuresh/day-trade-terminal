function doGet(e) {
  var p = (e && e.parameter) || {};
  var ffText = getFF_();
  var body = ffText;
  if (p.v === '2') {
    var ff = [];
    try { ff = JSON.parse(ffText); } catch (x) {}
    var actuals = {};
    try { actuals = getActuals_(ff); } catch (x) {}
    body = JSON.stringify({ v: 2, ff: ff, actuals: actuals, fred: !!FRED_KEY, at: new Date().toISOString() });
  }
  var cb = p.callback;
  if (cb && /^[A-Za-z_$][\w$]*$/.test(cb)) {
    return ContentService.createTextOutput(cb + '(' + body + ');')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  return ContentService.createTextOutput(body).setMimeType(ContentService.MimeType.JSON);
}

function getFF_() {
  var cache = CacheService.getScriptCache();
  var body = cache.get('ff');
  if (body) return body;
  try {
    var r = UrlFetchApp.fetch(FF_URL, { muteHttpExceptions: true });
    var t = r.getContentText();
    if (r.getResponseCode() === 200 && t.trim().charAt(0) === '[') {
      cache.put('ff', t, 1800);
      cache.put('ffgood', t, 21600);
      return t;
    }
  } catch (err) {}
  return cache.get('ffgood') || '[]';
}

function ruleFor_(ev) {
  for (var i = 0; i < RULES.length; i++) {
    if (RULES[i][0] === ev.country && RULES[i][1] === ev.title) return RULES[i];
  }
  return null;
}

function getActuals_(ff) {
  var props = PropertiesService.getScriptProperties();
  var store = {};
  try { store = JSON.parse(props.getProperty('actuals') || '{}'); } catch (x) {}
  var now = Date.now(), out = {}, need = {};
  ff.forEach(function (ev) {
    var rule = ruleFor_(ev);
    if (!rule) return;
    var t = Date.parse(ev.date);
    if (!t || t > now || now - t > 8 * 864e5) return;   // only releases already out this week
    var key = ev.country + '|' + ev.title + '|' + ev.date;
    if (store[key]) { out[key] = store[key]; return; }
    (need[rule[2]] = need[rule[2]] || []).push({ ev: ev, rule: rule, key: key, t: t });
  });
  var srcs = Object.keys(need);
  if (!srcs.length) return out;
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) return out;
  try {
    var cache = CacheService.getScriptCache();
    srcs.forEach(function (src) {
      if (cache.get('try_' + src)) return;              // asked under a minute ago
      cache.put('try_' + src, '1', 55);
      var got = {};
      try { got = FETCH[src](need[src]) || {}; } catch (x) {}
      Object.keys(got).forEach(function (k) { store[k] = got[k]; out[k] = got[k]; });
    });
    Object.keys(store).forEach(function (k) {
      var d = Date.parse(k.split('|').pop());
      if (!d || now - d > 10 * 864e5) delete store[k];
    });
    props.setProperty('actuals', JSON.stringify(store));
  } finally { lock.releaseLock(); }
  return out;
}

var FETCH = {};

// US Bureau of Labor Statistics: CPI, jobs report, PPI, JOLTS
FETCH.bls = function (items) {
  var ids = {};
  items.forEach(function (it) { ids[it.rule[3]] = 1; });
  var y = new Date().getFullYear();
  var payload = { seriesid: Object.keys(ids), startyear: String(y - 2), endyear: String(y) };
  if (BLS_KEY) payload.registrationkey = BLS_KEY;
  var url = 'https://api.bls.gov/publicAPI/' + (BLS_KEY ? 'v2' : 'v1') + '/timeseries/data/';
  var r = UrlFetchApp.fetch(url, { method: 'post', contentType: 'application/json', payload: JSON.stringify(payload), muteHttpExceptions: true });
  var j = JSON.parse(r.getContentText());
  if (j.status !== 'REQUEST_SUCCEEDED' || !j.Results) return {};
  var series = {};
  (j.Results.series || []).forEach(function (s) {
    var m = {};
    (s.data || []).forEach(function (d) {
      if (/^M(0[1-9]|1[0-2])$/.test(d.period)) {
        var v = parseFloat(String(d.value).replace(/,/g, ''));
        if (!isNaN(v)) m[d.year + '-' + d.period.slice(1)] = v;
      }
    });
    series[s.seriesID] = m;
  });
  var got = {};
  items.forEach(function (it) {
    var per = refMonth_(it.t, it.rule[5]);
    var m = series[it.rule[3]];
    var v = calc_(m, per, it.rule[4]);
    if (v === null) return;                              // not published yet
    got[it.key] = rec_(v, calc_(m, addMonths_(per, -1), it.rule[4]), it.rule[6], 'BLS', per, it.rule[7]);
  });
  return got;
};

// Federal Reserve: FOMC statement, same day
FETCH.fed = function (items) {
  var got = {};
  items.forEach(function (it) {
    var d = Utilities.formatDate(new Date(it.t), 'America/New_York', 'yyyyMMdd');
    var url = 'https://www.federalreserve.gov/newsevents/pressreleases/monetary' + d + 'a.htm';
    var r = UrlFetchApp.fetch(url, { muteHttpExceptions: true });
    if (r.getResponseCode() !== 200) return;
    var v = parseFed_(text_(r.getContentText()));
    if (v !== null) got[it.key] = rec_(v, null, '%', 'Federal Reserve', d, url);
  });
  return got;
};
function parseFed_(txt) {
  var m = txt.match(/target range for the federal funds rate(?: by [\d\/\-]+ percentage points?)? (?:at|to) (\d+(?:-\d\/\d)?) to (\d+(?:-\d\/\d)?) percent/i);
  return m ? frac_(m[2]) : null;
}
function frac_(s) {
  var m = String(s).match(/^(\d+)(?:-(\d)\/(\d))?$/);
  return m ? +m[1] + (m[2] ? +m[2] / +m[3] : 0) : null;
}

// Bank of England: Monetary Policy Summary, same day
FETCH.boe = function (items) {
  var got = {};
  items.forEach(function (it) {
    var d = new Date(it.t);
    var y = Utilities.formatDate(d, 'Europe/London', 'yyyy');
    var mon = Utilities.formatDate(d, 'Europe/London', 'MMMM').toLowerCase();
    var url = 'https://www.bankofengland.co.uk/monetary-policy-summary-and-minutes/' + y + '/' + mon + '-' + y;
    var r = UrlFetchApp.fetch(url, { muteHttpExceptions: true });
    if (r.getResponseCode() !== 200) return;
    var v = parseBoe_(text_(r.getContentText()));
    if (v !== null) got[it.key] = rec_(v, null, '%', 'Bank of England', y + '-' + mon, url);
  });
  return got;
};
function parseBoe_(txt) {
  var m = txt.match(/voted[^.]{0,160}?\bto (?:maintain|keep|leave|hold|reduce|cut|lower|increase|raise) Bank Rate(?: unchanged)?(?: by \d+(?:\.\d+)? percentage points?,?)? (?:at|to|unchanged at) (\d+(?:\.\d+)?)%/i);
  return m ? parseFloat(m[1]) : null;
}

// European Central Bank: monetary policy decision press release, same day
FETCH.ecb = function (items) {
  var rss = UrlFetchApp.fetch('https://www.ecb.europa.eu/rss/press.html', { muteHttpExceptions: true });
  if (rss.getResponseCode() !== 200) return {};
  var xml = rss.getContentText(), got = {}, pages = {};
  items.forEach(function (it) {
    var day = Utilities.formatDate(new Date(it.t), 'Europe/Berlin', 'yyMMdd');
    var link = null, re = /<item[\s\S]*?<\/item>/g, m;
    while ((m = re.exec(xml))) {
      var l = m[0].match(/<link>\s*([^<\s]+)\s*<\/link>/);
      if (l && l[1].indexOf('ecb.mp' + day) >= 0) { link = l[1]; break; }
    }
    if (!link) return;
    if (link.indexOf('http') !== 0) link = 'https://www.ecb.europa.eu' + link;
    if (!pages[link]) {
      var r = UrlFetchApp.fetch(link, { muteHttpExceptions: true });
      pages[link] = r.getResponseCode() === 200 ? text_(r.getContentText()) : '';
    }
    var rates = parseEcb_(pages[link]);
    if (rates && rates[it.rule[3]] != null) got[it.key] = rec_(rates[it.rule[3]], null, '%', 'ECB', day, link);
  });
  return got;
};
function parseEcb_(txt) {
  var m = txt.match(/interest rates on the ([a-z ,]+?) will (?:be |remain )?(?:unchanged at|decreased to|increased to|reduced to|lowered to|raised to|kept at|at) (\d+(?:\.\d+)?)%, (\d+(?:\.\d+)?)% and (\d+(?:\.\d+)?)%/i);
  if (!m) return null;
  var names = m[1].toLowerCase(), vals = [parseFloat(m[2]), parseFloat(m[3]), parseFloat(m[4])];
  var order = [['dfr', names.indexOf('deposit')], ['mro', names.indexOf('main refinancing')], ['mlf', names.indexOf('marginal lending')]]
    .filter(function (x) { return x[1] >= 0; }).sort(function (a, b) { return a[1] - b[1]; });
  if (order.length !== 3) return null;
  var out = {};
  order.forEach(function (x, i) { out[x[0]] = vals[i]; });
  return out;
}

// UK Office for National Statistics: CPI time series, same day
FETCH.ons = function (items) {
  var got = {};
  var MONTHS = ['January', 'February', 'March', 'April', 'May', 'June', 'July', 'August', 'September', 'October', 'November', 'December'];
  items.forEach(function (it) {
    var page = 'https://www.ons.gov.uk/economy/inflationandpriceindices/timeseries/' + it.rule[3] + '/mm23';
    var r = UrlFetchApp.fetch(page + '/data', { muteHttpExceptions: true });
    if (r.getResponseCode() !== 200) return;
    var j = JSON.parse(r.getContentText()), m = {};
    (j.months || []).forEach(function (x) {
      var mi = MONTHS.indexOf(x.month);
      var v = parseFloat(x.value);
      if (mi >= 0 && !isNaN(v)) m[x.year + '-' + ('0' + (mi + 1)).slice(-2)] = v;
    });
    var per = refMonth_(it.t, it.rule[5]);
    if (m[per] === undefined) return;
    got[it.key] = rec_(m[per], m[addMonths_(per, -1)] === undefined ? null : m[addMonths_(per, -1)], '%', 'ONS', per, page);
  });
  return got;
};

// Eurostat: euro area HICP (flash estimate and final), annual rate
FETCH.eurostat = function (items) {
  var got = {}, cache = {};
  var CODES = { all: ['CP00', 'TOTAL'], core: ['TOT_X_NRG_FOOD', 'TOT_X_NRG_FOOD_NP'] };
  items.forEach(function (it) {
    var per = refMonth_(it.t, it.rule[5]), m = null, used = '';
    var sets = ['prc_hicp_minr', 'prc_hicp_manr'], codes = CODES[it.rule[3]];
    for (var a = 0; a < sets.length && !m; a++) for (var b = 0; b < codes.length && !m; b++) {
      var url = 'https://ec.europa.eu/eurostat/api/dissemination/statistics/1.0/data/' + sets[a] + '?geo=EA&unit=RCH_A&coicop=' + codes[b] + '&lastTimePeriod=3';
      if (cache[url] === undefined) {
        var r = UrlFetchApp.fetch(url, { muteHttpExceptions: true });
        cache[url] = r.getResponseCode() === 200 ? jsonStat_(r.getContentText()) : null;
      }
      if (cache[url] && cache[url][per] !== undefined) { m = cache[url]; used = url; }
    }
    if (!m) return;
    var pv = m[addMonths_(per, -1)];
    got[it.key] = rec_(m[per], pv === undefined ? null : pv, '%', 'Eurostat', per, 'https://ec.europa.eu/eurostat/web/hicp');
  });
  return got;
};
function jsonStat_(txt) {
  var j = JSON.parse(txt), out = {};
  var tIdx = j.dimension && j.dimension.time && j.dimension.time.category && j.dimension.time.category.index;
  if (!tIdx || !j.value) return out;
  Object.keys(tIdx).forEach(function (t) {
    var v = j.value[tIdx[t]];
    var ym = String(t).replace(/^(\d{4})M(\d{2})$/, '$1-$2');
    if (v !== undefined && v !== null) out[ym] = +v;
  });
  return out;
}

// FRED (Federal Reserve Bank of St. Louis) for jobless claims, PCE, retail sales and GDP. Needs FRED_KEY.
FETCH.fred = function (items) {
  if (!FRED_KEY) return {};
  var got = {}, series = {};
  items.forEach(function (it) {
    var id = it.rule[3];
    if (!series[id]) {
      var url = 'https://api.stlouisfed.org/fred/series/observations?series_id=' + id + '&api_key=' + FRED_KEY + '&file_type=json&sort_order=desc&limit=40';
      var r = UrlFetchApp.fetch(url, { muteHttpExceptions: true });
      var m = {};
      if (r.getResponseCode() === 200) {
        (JSON.parse(r.getContentText()).observations || []).forEach(function (o) {
          var v = parseFloat(o.value);
          if (!isNaN(v)) m[o.date] = v;
        });
      }
      series[id] = m;
    }
    var m2 = series[id], how = it.rule[4], v = null, pv = null, per = '';
    if (how === 'weekK') {
      // claims cover the week ending the Saturday before the release
      var dow = +Utilities.formatDate(new Date(it.t), 'America/New_York', 'u');   // 1 = Monday … 7 = Sunday
      var back = ((dow - 6 + 7) % 7) || 7;
      per = Utilities.formatDate(new Date(it.t - back * 864e5), 'America/New_York', 'yyyy-MM-dd');
      var prevWeek = Utilities.formatDate(new Date(it.t - (back + 7) * 864e5), 'America/New_York', 'yyyy-MM-dd');
      if (m2[per] !== undefined) { v = Math.round(m2[per] / 1000); pv = m2[prevWeek] !== undefined ? Math.round(m2[prevWeek] / 1000) : null; }
    } else if (how === 'gdp') {
      var ym = Utilities.formatDate(new Date(it.t), 'America/New_York', 'yyyy-MM');
      var y = +ym.slice(0, 4), mo = +ym.slice(5, 7);
      var q = mo <= 3 ? (y - 1) + '-10' : mo <= 6 ? y + '-01' : mo <= 9 ? y + '-04' : y + '-07';
      per = q + '-01';
      if (m2[per] !== undefined) v = round_(m2[per], 1);
    } else {
      var mm = {};
      Object.keys(m2).forEach(function (d) { mm[d.slice(0, 7)] = m2[d]; });
      per = refMonth_(it.t, it.rule[5]);
      v = calc_(mm, per, how);
      pv = calc_(mm, addMonths_(per, -1), how);
    }
    if (v !== null) got[it.key] = rec_(v, pv, it.rule[6], 'FRED', per, it.rule[7]);
  });
  return got;
};

// ---- helpers ----
function refMonth_(t, back) {
  var ym = Utilities.formatDate(new Date(t), 'America/New_York', 'yyyy-MM');
  return addMonths_(ym, -back);
}
function addMonths_(ym, n) {
  var y = +ym.slice(0, 4), m = +ym.slice(5, 7) - 1 + n;
  y += Math.floor(m / 12); m = ((m % 12) + 12) % 12;
  return y + '-' + ('0' + (m + 1)).slice(-2);
}
function round_(x, d) { var f = Math.pow(10, d); return Math.round(x * f + (x >= 0 ? 1e-7 : -1e-7)) / f; }
function calc_(m, per, how) {
  if (!m || m[per] === undefined) return null;
  var v0 = m[per], p1 = m[addMonths_(per, -1)], p12 = m[addMonths_(per, -12)];
  if (how === 'level') return round_(v0, 1);
  if (how === 'levelM') return round_(v0 / 1000, 2);
  if (how === 'pct1') return p1 ? round_((v0 / p1 - 1) * 100, 1) : null;
  if (how === 'pct12') return p12 ? round_((v0 / p12 - 1) * 100, 1) : null;
  if (how === 'diff1') return p1 !== undefined ? Math.round(v0 - p1) : null;
  return null;
}
function rec_(v, pv, unit, src, per, url) {
  return { v: v, pv: pv, u: unit, src: src, per: per, url: url || '', at: new Date().toISOString() };
}
function text_(html) {
  return String(html).replace(/&#8209;|&#x2011;|[\u2010-\u2015\u2212]/g, '-')
    .replace(/\s*-?\s*½/g, '-1/2').replace(/\s*-?\s*¼/g, '-1/4').replace(/\s*-?\s*¾/g, '-3/4').replace(/<script[\s\S]*?<\/script>/gi, ' ').replace(/<style[\s\S]*?<\/style>/gi, ' ')
    .replace(/<[^>]+>/g, ' ').replace(/&nbsp;|&#160;/g, ' ').replace(/&amp;/g, '&').replace(/&#8211;|&ndash;/g, '–')
    .replace(/\s+/g, ' ');
}
